ดึงรายชื่อนักเรียนทั้งหมดมาเก็บใน ESP32
C++ / Arduino
// ── โครงสร้างข้อมูลนักเรียนใน RAM ──────────────────
struct Student {
  String name;
  String className;
  bool   active;
};

// UID → Student map (O(1) lookup)
std::unordered_map<std::string, Student> uidCache;

// ── ดึง UID list จาก Google Apps Script ─────────────
bool fetchAndCacheUIDs() {
  HTTPClient http;
  http.begin(SERVER_URL + "/api/uids");
  http.addHeader("X-Device-Key", DEVICE_KEY);
  
  int code = http.GET();
  if (code != 200) return false;
  
  String payload = http.getString();
  http.end();

  // ── Parse JSON → HashMap ──────────────────────────
  DynamicJsonDocument doc(32768); // 32KB สำหรับ 500 รายการ
  deserializeJson(doc, payload);
  
  uidCache.clear();
  for (auto entry : doc["students"].as<JsonArray>()) {
    Student s;
    s.name      = entry["name"].as<String>();
    s.className = entry["class"].as<String>();
    s.active    = entry["active"].as<bool>();
    uidCache[entry["uid"].as<std::string>()] = s;
  }
  
  // ── บันทึกลง SPIFFS สำหรับ offline ──────────────
  writeFile("/uid_cache.json", payload);
  lastSyncTime = millis();
  
  Serial.printf("Cache: %d UIDs loaded\n", uidCache.size());
  return true;
}

// ── Periodic re-sync ทุก 6 ชั่วโมง ──────────────────
void checkCacheExpiry() {
  if (millis() - lastSyncTime > 21600000UL) { // 6h
    fetchAndCacheUIDs();
  }
}
Response JSON Format จาก Apps Script ควรมีโครงสร้าง: {"students":[{"uid":"A1B2C3D4","name":"สมชาย ใจดี","class":"ปวช.1/1","active":true}...],"updated":"2025-01-01T00:00:00"}
07
ระบบ Cache เพื่อลดความหน่วง
ตอบสนองภายใน 50ms แม้ไม่มี internet
Cache Architecture (3 ชั้น)
L1
RAM HashMap — <1ms, หายเมื่อ reboot
L2
SPIFFS File — ~20ms, คงอยู่หลัง reboot
L3
Google Sheets — 500-2000ms, source of truth
Response Time เปรียบเทียบ
🟢 Cache Hit (RAM): < 1 ms
🔵 Cache Hit (SPIFFS): 10–30 ms
🌐 API Call (WiFi): 300–800 ms
❌ API Timeout: 5000 ms
🎯 เป้าหมาย: 95%+ ของการสแกน < 50ms
C++ / Cache Lookup Logic
// ── ค้นหาแบบ cascading cache ──────────────────────────
ScanResult lookupUID(const String& uid) {
  ScanResult result;
  
  // L1: Check RAM HashMap first (fastest)
  auto it = uidCache.find(uid.c_str());
  if (it != uidCache.end()) {
    result.found  = true;
    result.name   = it->second.name;
    result.klass  = it->second.className;
    result.active = it->second.active;
    result.source = "RAM";
    return result;
  }
  
  // L3: Online fallback (only if WiFi connected)
  if (WiFi.status() == WL_CONNECTED) {
    result = queryServerForUID(uid); // HTTP GET
    if (result.found) {
      // Add to L1 cache dynamically
      uidCache[uid.c_str()] = {result.name, result.klass, true};
    }
  }
  
  return result;
}

// ── FreeRTOS Task: ส่ง log ไป server ใน background ──
void serverTask(void* param) {
  ScanRecord rec;
  while(true) {
    if (xQueueReceive(scanQueue, &rec, portMAX_DELAY)) {
      if (WiFi.status() == WL_CONNECTED) {
        postAttendance(rec);     // POST ไป Apps Script
      } else {
        saveOfflineQueue(rec);  // เก็บใน SPIFFS
      }
    }
  }
}

// ── เปิด AP + Web Server ──────────────────────────────
void startAPMode() {
  String apName = "ESP32-RFID-" + getMacSuffix();
  WiFi.softAP(apName, "12345678");  // รหัสเริ่มต้น
  
  displayAPScreen(WiFi.softAPIP().toString());
  
  // ── Routes ────────────────────────────────────────
  server.on("/", HTTP_GET, [](AsyncWebServerRequest* req) {
    req->send_P(200, "text/html", WIFI_CONFIG_HTML);
  });
  
  server.on("/save", HTTP_POST, [](AsyncWebServerRequest* req) {
    String ssid = req->getParam("ssid", true)->value();
    String pass = req->getParam("pass", true)->value();
    
    // บันทึกลง SPIFFS
    DynamicJsonDocument doc(256);
    doc["ssid"] = ssid; doc["pass"] = pass;
    writeJsonFile("/wifi.json", doc);
    
    req->send(200, "text/plain", "Saved! Rebooting...");
    delay(1000);
    ESP.restart();
  });
  
  server.on("/status", HTTP_GET, [](AsyncWebServerRequest* req) {
    DynamicJsonDocument doc(512);
    doc["ip"]         = WiFi.localIP().toString();
    doc["rssi"]       = WiFi.RSSI();
    doc["cacheCount"] = (int)uidCache.size();
    doc["freeHeap"]   = ESP.getFreeHeap();
    String out; serializeJson(doc, out);
    req->send(200, "application/json", out);
  });
  
  server.begin();
  apModeActive = true;
}

function doPost(e) {
  const data   = JSON.parse(e.postData.contents);
  const action = e.parameter.action || data.action;
  
  if (action === 'attendance') {
    const ss    = SpreadsheetApp.openById(SHEET_ID);
    const sheet = ss.getSheetByName('attendance');
    sheet.appendRow([
      data.timestamp, data.uid, data.name,
      data.class, data.device, data.result
    ]);
    return jsonResponse({ok: true});
  }
  
  if (action === 'batch') {
    const rows = data.records.map(r => [
      r.timestamp, r.uid, r.name, r.class, r.device, r.result
    ]);
    sheet.getRange(sheet.getLastRow()+1, 1, rows.length, 6)
         .setValues(rows);
    return jsonResponse({ok: true, count: rows.length});
  }
}

function doGet(e) {
  if (e.parameter.action === 'uids') {
    const data = ss.getSheetByName('students')
                    .getDataRange().getValues();
    const students = data.slice(1).map(row => ({
      uid: row[0], name: row[1], class: row[2], active: row[3]
    }));
    return jsonResponse({students, updated: new Date().toISOString()});
  }
}

// ── Task Handles ──────────────────────────────────────
TaskHandle_t hScan, hServer, hDisplay;
QueueHandle_t scanQueue;
SemaphoreHandle_t displayMutex;

void setup() {
  initHardware();      // SPI, I2C, GPIO, Serial
  initSPIFFS();        // Mount filesystem
  loadConfig();        // config.json
  showBootScreen();    // "BOOTING..."
  
  // ── WiFi ─────────────────────────────────────────
  if (!connectWiFi()) {
    if (hasCacheFile()) {
      loadCacheFromSPIFFS();
      mode = OFFLINE_MODE;
    } else {
      startAPMode();
      return;  // loop() จะ handle AP mode
    }
  } else {
    syncNTP();
    fetchAndCacheUIDs();
    mode = ONLINE_MODE;
  }
  
  // ── FreeRTOS Queues & Mutexes ──────────────────
  scanQueue    = xQueueCreate(20, sizeof(ScanRecord));
  displayMutex = xSemaphoreCreateMutex();
  
  // ── Start Tasks (Core assignments) ─────────────
  xTaskCreatePinnedToCore(scanTask,    "scan",    4096, nullptr, 3, &hScan,    1);
  xTaskCreatePinnedToCore(serverTask,  "server",  8192, nullptr, 1, &hServer,  0);
  xTaskCreatePinnedToCore(displayTask, "display", 4096, nullptr, 2, &hDisplay, 1);
}

void loop() {
  // loop() ว่างเกือบหมด — งานอยู่ใน Tasks
  // ใช้แค่ตรวจ AP config button
  if (digitalRead(CONFIG_BTN) == LOW) {
    configBtnPressTime = millis();
    while(digitalRead(CONFIG_BTN) == LOW);
    if (millis() - configBtnPressTime > 5000) startAPMode();
  }
  vTaskDelay(pdMS_TO_TICKS(100));
}

// ── Task Priorities ───────────────────────────────────
// scanTask   → Core 1, Priority 3 (highest — response time สำคัญ)
// displayTask → Core 1, Priority 2
// serverTask  → Core 0, Priority 1 (network I/O แยก core)

ปัญหาที่พบบ่อยและวิธีแก้
จากประสบการณ์ deploy จริงในโรงเรียน
ไฟตก / ESP32 รีสตาร์ทกะทันหัน (Brownout Reset)
สาเหตุ: MFRC522 ดูดกระแสตอนส่งสัญญาณ RF ทำให้ไฟ 3.3V วูบ
วิธีแก้:
  • เพิ่ม capacitor 100µF (electrolytic) ระหว่าง 3.3V กับ GND ใกล้ MFRC522
  • เพิ่ม 100nF ceramic cap คู่กัน (bypass HF noise)
  • ใช้ power adapter 5V 2A คุณภาพดี ไม่ใช่ charger มือถือ
  • ตั้ง Brownout threshold: WRITE_PERI_REG(RTC_CNTL_BROWN_OUT_REG, 0) (ระวัง: อาจทำให้ chip เสียหายถ้าไฟต่ำจริงๆ)
RAM เต็ม / Heap Overflow / Task Stack Overflow
สาเหตุ: ArduinoJson DynamicJsonDocument ขนาดใหญ่, String concatenation, UID cache โตเกิน
วิธีแก้:
  • ใช้ Serial.println(ESP.getFreeHeap()) monitor heap ตลอด
  • ใช้ StaticJsonDocument แทน Dynamic เมื่อรู้ขนาดแน่นอน
  • จำกัด cache ไม่เกิน 500 UID (ประมาณ 24KB heap)
  • เพิ่ม Task Stack: ถ้า crash → เพิ่ม stack size ใน xTaskCreate
  • ใช้ ESP32-S3 ที่มี PSRAM 4MB สำหรับ cache ขนาดใหญ่
WiFi หลุดบ่อย / Reconnect loop
สาเหตุ: สัญญาณอ่อน, DHCP timeout, router บล็อก
วิธีแก้:
  • ตั้ง Static IP แทน DHCP: WiFi.config(ip, gateway, subnet)
  • ใช้ WiFi Event Callback แทน while loop polling
  • เปิด Auto-reconnect: WiFi.setAutoReconnect(true)
  • ถ้า WiFi หลุด → ทำงาน Offline Mode ต่อ อย่า block scanTask
NTP เวลาผิด / Timestamp ผิดพลาด
สาเหตุ: ต่อ timezone ผิด, sync ไม่สำเร็จ, RTC drift หลัง reboot
วิธีแก้:
  • ตั้ง timezone ไทย: configTime(7*3600, 0, "pool.ntp.org", "time.google.com")
  • รอ NTP sync จริงๆ: while(!time(nullptr)) delay(100)
  • เพิ่ม DS3231 RTC module สำหรับเวลาแม่นยำแม้ไม่มี internet
อ่านการ์ดได้ไม่สม่ำเสมอ / Double scan
สาเหตุ: ไม่ได้ HALT การ์ดหลังอ่าน, debounce ไม่ดี, antenna gain ต่ำ
วิธีแก้:
  • เรียก mfrc522.PICC_HaltA() และ mfrc522.PCD_StopCrypto1() หลังอ่านทุกครั้ง
  • ใช้ debounce 3 วินาทีต่อ UID เดียวกัน
  • ปรับ antenna gain: mfrc522.PCD_SetAntennaGain(0x70)
  • วางเสาอากาศห่างจากวัตถุโลหะ
SPIFFS เขียนบ่อยเกิน → Flash เสื่อม
สาเหตุ: เขียน scan log ลง SPIFFS ทุกครั้งที่สแกน (Flash อายุ ~100,000 write cycles)
วิธีแก้:
  • เขียน offline queue ลง SPIFFS เฉพาะเมื่อ WiFi หลุดเท่านั้น
  • Batch write แทนการเขียนทีละ record
  • ใช้ LittleFS แทน SPIFFS (wear leveling ดีกว่า)
  • Cache file (uid_cache.json) sync แค่วันละ 2 ครั้ง
